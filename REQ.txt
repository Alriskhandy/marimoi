    public function index(Request $request)
    {
        $query = Aspirasi::with(['kategoriAspirasi.opd', 'admin']);

        // Filter by status
        if ($request->has('status') && $request->status != '') {
            $query->where('status', $request->status);
        }

        // Filter by priority
        if ($request->has('prioritas') && $request->prioritas != '') {
            $query->where('prioritas', $request->prioritas);
        }

        // Filter by category
        if ($request->has('kategori') && $request->kategori != '') {
            $query->where('kategori_aspirasi_id', $request->kategori);
        }

        // Filter by date range
        if ($request->has('tanggal_dari') && $request->tanggal_dari != '') {
            $query->whereDate('created_at', '>=', $request->tanggal_dari);
        }
        if ($request->has('tanggal_sampai') && $request->tanggal_sampai != '') {
            $query->whereDate('created_at', '<=', $request->tanggal_sampai);
        }

        // Search
        if ($request->has('search') && $request->search != '') {
            $search = $request->search;
            $query->where(function($q) use ($search) {
                $q->where('nomor_tiket', 'like', "%{$search}%")
                  ->orWhere('nama_pengirim', 'like', "%{$search}%")
                  ->orWhere('judul_aspirasi', 'like', "%{$search}%")
                  ->orWhere('isi_aspirasi', 'like', "%{$search}%");
            });
        }

        $aspirasi = $query->orderBy('created_at', 'desc')->paginate(15);
        
        // Data for filters
        $categories = KategoriAspirasi::active()->get();
        $opds = Opd::active()->get();

        return view('aspirasi.index', compact('aspirasi', 'categories', 'opds'));
    }

    public function create()
    {
        $categories = KategoriAspirasi::with('opd')->active()->get();
        $opds = Opd::active()->get();
        return view('aspirasi.create', compact('categories', 'opds'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'kategori_aspirasi_id' => 'required|exists:kategori_aspirasis,id',
            'nama_pengirim' => 'required|string|max:255',
            'email' => 'required|email',
            'phone' => 'nullable|string|max:20',
            'alamat' => 'required|string',
            'jenis_aspirasi' => 'required|in:usulan,keluhan,kritik,saran',
            'judul_aspirasi' => 'required|string|max:255',
            'isi_aspirasi' => 'required|string',
            'lampiran.*' => 'nullable|file|mimes:pdf,doc,docx,jpg,jpeg,png|max:5120',
            'latitude' => 'nullable|numeric|between:-90,90',
            'longitude' => 'nullable|numeric|between:-180,180',
            'prioritas' => 'required|in:rendah,sedang,tinggi,urgent'
        ]);

        $data = $request->all();

        // Handle file uploads
        if ($request->hasFile('lampiran')) {
            $lampiran = [];
            foreach ($request->file('lampiran') as $file) {
                $filename = time() . '_' . $file->getClientOriginalName();
                $path = $file->storeAs('aspirasi-lampiran', $filename, 'public');
                $lampiran[] = [
                    'filename' => $file->getClientOriginalName(),
                    'path' => $path,
                    'size' => $file->getSize(),
                    'mime_type' => $file->getMimeType()
                ];
            }
            $data['lampiran'] = $lampiran;
        }

        Aspirasi::create($data);

        return redirect()->route('aspirasi.index')
                         ->with('success', 'Aspirasi berhasil ditambahkan.');
    }

    public function show(Aspirasi $aspirasi)

        return view('aspirasi.show', compact('aspirasi'));
    }

        return view('aspirasi.edit', compact('aspirasi', 'categories', 'opds'));
    }

    public function update(Request $request, Aspirasi $aspirasi)
    {
        $request->validate([
            'kategori_aspirasi_id' => 'required|exists:kategori_aspirasis,id',
            'nama_pengirim' => 'required|string|max:255',
            'email' => 'required|email',
            'phone' => 'nullable|string|max:20',
            'alamat' => 'required|string',
            'jenis_aspirasi' => 'required|in:usulan,keluhan,kritik,saran',
            'judul_aspirasi' => 'required|string|max:255',
            'isi_aspirasi' => 'required|string',
            'lampiran.*' => 'nullable|file|mimes:pdf,doc,docx,jpg,jpeg,png|max:5120',
            'latitude' => 'nullable|numeric|between:-90,90',
            'longitude' => 'nullable|numeric|between:-180,180',
            'prioritas' => 'required|in:rendah,sedang,tinggi,urgent'
        ]);

        $data = $request->all();

        // Handle new file uploads
        if ($request->hasFile('lampiran')) {
            // Delete old files if needed
            if ($aspirasi->lampiran) {
                foreach ($aspirasi->lampiran as $file) {
                    Storage::disk('public')->delete($file['path']);
                }
            }

            $lampiran = [];
            foreach ($request->file('lampiran') as $file) {
                $filename = time() . '_' . $file->getClientOriginalName();
                $path = $file->storeAs('aspirasi-lampiran', $filename, 'public');
                $lampiran[] = [
                    'filename' => $file->getClientOriginalName(),
                    'path' => $path,
                    'size' => $file->getSize(),
                    'mime_type' => $file->getMimeType()
                ];
            }
            $data['lampiran'] = $lampiran;
        }

        $aspirasi->update($data);

        return redirect()->route('aspirasi.index')
                         ->with('success', 'Aspirasi berhasil diperbarui.');
    }

    public function destroy(Aspirasi $aspirasi)
    {
        try {
            // Delete associated files
            if ($aspirasi->lampiran) {
                foreach ($aspirasi->lampiran as $file) {
                    Storage::disk('public')->delete($file['path']);
                }
            }

            $aspirasi->delete();
            return redirect()->route('aspirasi.index')
                             ->with('success', 'Aspirasi berhasil dihapus.');
        } catch (\Exception $e) {
            return redirect()->route('aspirasi.index')
                             ->with('error', 'Terjadi kesalahan saat menghapus aspirasi.');
        }
    }

    // Method untuk update status
    public function updateStatus(Request $request, Aspirasi $aspirasi)
    {
        $request->validate([
            'status' => 'required|in:pending,diproses,selesai,ditolak',
            'tanggapan_admin' => 'nullable|string'
        ]);

        $aspirasi->updateStatus(
            $request->status,
            auth()->id(),
            $request->tanggapan_admin
        );

        return redirect()->back()
                         ->with('success', 'Status aspirasi berhasil diperbarui.');
    }

    // Method untuk dashboard/statistik
    public function dashboard()
    {
        $stats = [
            'total' => Aspirasi::count(),
            'pending' => Aspirasi::pending()->count(),
            'diproses' => Aspirasi::diproses()->count(),
            'selesai' => Aspirasi::selesai()->count(),
            'bulan_ini' => Aspirasi::whereMonth('created_at', Carbon::now()->month)
                                 ->whereYear('created_at', Carbon::now()->year)
                                 ->count(),
        ];

        // Data untuk chart
        $chartData = Aspirasi::selectRaw('DATE(created_at) as tanggal, COUNT(*) as jumlah')
                           ->whereDate('created_at', '>=', Carbon::now()->subDays(30))
                           ->groupBy('tanggal')
                           ->orderBy('tanggal')
                           ->get();

        // Data berdasarkan kategori
        $byCategory = Aspirasi::with('kategoriAspirasi')
                            ->selectRaw('kategori_aspirasi_id, COUNT(*) as jumlah')
                            ->groupBy('kategori_aspirasi_id')
                            ->orderBy('jumlah', 'desc')
                            ->limit(5)
                            ->get();

        return view('aspirasi.dashboard', compact('stats', 'chartData', 'byCategory'));
    }

    // Method untuk download lampiran
    public function downloadLampiran(Aspirasi $aspirasi, $index)
    {
        if (!isset($aspirasi->lampiran[$index])) {
            abort(404);
        }

        $file = $aspirasi->lampiran[$index];
        $filePath = storage_path('app/public/' . $file['path']);

        if (!file_exists($filePath)) {
            abort(404);
        }

        return response()->download($filePath, $file['filename']);
    }

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Carbon\Carbon;

class Aspirasi extends Model
{
    use HasFactory;

    protected $fillable = [
        'kategori_aspirasi_id',
        'nomor_tiket',
        'nama_pengirim',
        'email',
        'phone',
        'alamat',
        'jenis_aspirasi',
        'judul_aspirasi',
        'isi_aspirasi',
        'lampiran',
        'latitude',
        'longitude',
        'tanggapan_admin',
        'status',
        'prioritas',
        'tanggal_respon',
        'admin_id'
    ];

    protected $casts = [
        'lampiran' => 'array',
        'latitude' => 'decimal:8',
        'longitude' => 'decimal:8',
        'tanggal_respon' => 'datetime',
    ];

    // Boot method for auto-generating ticket number
    protected static function boot()
    {
        parent::boot();

        static::creating(function ($aspirasi) {
            if (empty($aspirasi->nomor_tiket)) {
                $aspirasi->nomor_tiket = $aspirasi->generateTicketNumber();
            }
        });
    }

    // Relationships
    public function kategoriAspirasi()
    {
        return $this->belongsTo(KategoriAspirasi::class);
    }

    public function admin()
    {
        return $this->belongsTo(User::class, 'admin_id');
    }

    // Scopes
    public function scopePending($query)
    {
        return $query->where('status', 'pending');
    }

    public function scopeDiproses($query)
    {
        return $query->where('status', 'diproses');
    }

    public function scopeSelesai($query)
    {
        return $query->where('status', 'selesai');
    }

    public function scopeByPrioritas($query, $prioritas)
    {
        return $query->where('prioritas', $prioritas);
    }

    // Accessors
    public function getStatusBadgeAttribute()
    {
        $badges = [
            'pending' => 'warning',
            'diproses' => 'info',
            'selesai' => 'success',
            'ditolak' => 'danger',
        ];

        return $badges[$this->status] ?? 'secondary';
    }

    // Methods
    private function generateTicketNumber()
    {
        $date = Carbon::now()->format('Ymd');
        $lastTicket = static::whereDate('created_at', Carbon::today())
                          ->orderBy('id', 'desc')
                          ->first();
        
        $sequence = $lastTicket ? (int) substr($lastTicket->nomor_tiket, -4) + 1 : 1;
        
        return 'ASP' . $date . str_pad($sequence, 4, '0', STR_PAD_LEFT);
    }

    public function updateStatus($status, $adminId = null, $tanggapan = null)
    {
        $this->update([
            'status' => $status,
            'admin_id' => $adminId,
            'tanggapan_admin' => $tanggapan,
            'tanggal_respon' => now()
        ]);
    }
}

            'telepon' => 'nullable|string|max:20',
            'email' => 'nullable|email',
            'website' => 'nullable|url',
            'is_active' => 'boolean'
        ]);

        $data = $request->all();

        if ($request->hasFile('logo')) {
            $data['logo'] = $request->file('logo')->store('opd-logos', 'public');
        }

        Opd::create($data);

        return redirect()->route('opds.index')
                         ->with('success', 'OPD berhasil ditambahkan.');

        $opd->load('categoriesAspirasi', 'aspirasi');
        return view('opds.show', compact('opd'));

    {
        return view('opds.edit', compact('opd'));
    }

    public function update(Request $request, Opd $opd)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'singkatan' => 'required|string|max:20|unique:opds,singkatan,' . $opd->id,
            'logo' => 'nullable|image|mimes:jpeg,png,jpg|max:2048',
            'alamat' => 'nullable|string',
            'telepon' => 'nullable|string|max:20',
            'email' => 'nullable|email',
            'website' => 'nullable|url',
            'is_active' => 'boolean'
        ]);

        $data = $request->all();

        if ($request->hasFile('logo')) {
            if ($opd->logo) {
                Storage::disk('public')->delete($opd->logo);
            }
            $data['logo'] = $request->file('logo')->store('opd-logos', 'public');
        }

        $opd->update($data);

        return redirect()->route('opds.index')
                         ->with('success', 'OPD berhasil diperbarui.');
    }

    public function destroy(Opd $opd)
    {
        try {
            if ($opd->logo) {
                Storage::disk('public')->delete($opd->logo);
            }
            $opd->delete();
            return redirect()->route('opds.index')
                             ->with('success', 'OPD berhasil dihapus.');

            return redirect()->route('opds.index')
                             ->with('error', 'OPD tidak dapat dihapus karena masih memiliki data terkait.');

<?php

namespace App\Http\Controllers;

use App\Models\Aspirasi;
use App\Models\CategoriesAspirasi;
use App\Models\Opd;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Carbon\Carbon;

class AspirasiController extends Controller
{
    public function index(Request $request)
    {
        $query = Aspirasi::with(['categoriesAspirasi.opd', 'admin']);

        // Filter by status
        if ($request->has('status') && $request->status != '') {
            $query->where('status', $request->status);
        }

        // Filter by priority
        if ($request->has('prioritas') && $request->prioritas != '') {
            $query->where('prioritas', $request->prioritas);
        }

        // Filter by category
        if ($request->has('kategori') && $request->kategori != '') {
            $query->where('categories_aspirasi_id', $request->kategori);
        }

        // Filter by date range
        if ($request->has('tanggal_dari') && $request->tanggal_dari != '') {
            $query->whereDate('created_at', '>=', $request->tanggal_dari);
        }
        if ($request->has('tanggal_sampai') && $request->tanggal_sampai != '') {
            $query->whereDate('created_at', '<=', $request->tanggal_sampai);
        }

        // Search
        if ($request->has('search') && $request->search != '') {
            $search = $request->search;
            $query->where(function($q) use ($search) {
                $q->where('nomor_tiket', 'like', "%{$search}%")
                  ->orWhere('nama_pengirim', 'like', "%{$search}%")
                  ->orWhere('judul_aspirasi', 'like', "%{$search}%")
                  ->orWhere('isi_aspirasi', 'like', "%{$search}%");
            });
        }

        $aspirasi = $query->orderBy('created_at', 'desc')->paginate(15);
        
        // Data for filters
        $categories = CategoriesAspirasi::active()->get();
        $opds = Opd::active()->get();

        return view('aspirasi.index', compact('aspirasi', 'categories', 'opds'));
    }

    public function create()
    {
        $categories = CategoriesAspirasi::with('opd')->active()->get();
        $opds = Opd::active()->get();
        return view('aspirasi.create', compact('categories', 'opds'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'categories_aspirasi_id' => 'required|exists:categories_aspirasis,id',
            'nama_pengirim' => 'required|string|max:255',
            'email' => 'required|email',
            'phone' => 'nullable|string|max:20',
            'alamat' => 'required|string',
            'jenis_aspirasi' => 'required|in:usulan,keluhan,kritik,saran',
            'judul_aspirasi' => 'required|string|max:255',
            'isi_aspirasi' => 'required|string',
            'lampiran.*' => 'nullable|file|mimes:pdf,doc,docx,jpg,jpeg,png|max:5120',
            'latitude' => 'nullable|numeric|between:-90,90',
            'longitude' => 'nullable|numeric|between:-180,180',
            'prioritas' => 'required|in:rendah,sedang,tinggi,urgent'
        ]);

        $data = $request->all();

        // Handle file uploads
        if ($request->hasFile('lampiran')) {
            $lampiran = [];
            foreach ($request->file('lampiran') as $file) {
                $filename = time() . '_' . $file->getClientOriginalName();
                $path = $file->storeAs('aspirasi-lampiran', $filename, 'public');
                $lampiran[] = [
                    'filename' => $file->getClientOriginalName(),
                    'path' => $path,
                    'size' => $file->getSize(),
                    'mime_type' => $file->getMimeType()
                ];
            }
            $data['lampiran'] = $lampiran;
        }

        Aspirasi::create($data);

        return redirect()->route('aspirasi.index')
                         ->with('success', 'Aspirasi berhasil ditambahkan.');
    }

    public function show(Aspirasi $aspirasi)
    {
        $aspirasi->load(['categoriesAspirasi.opd', 'admin']);
        return view('aspirasi.show', compact('aspirasi'));

    {

        return view('aspirasi.edit', compact('aspirasi', 'categories', 'opds'));
    }

    public function update(Request $request, Aspirasi $aspirasi)
    {
        $request->validate([
            'categories_aspirasi_id' => 'required|exists:categories_aspirasis,id',
            'nama_pengirim' => 'required|string|max:255',
            'email' => 'required|email',
            'phone' => 'nullable|string|max:20',
            'alamat' => 'required|string',
            'jenis_aspirasi' => 'required|in:usulan,keluhan,kritik,saran',
            'judul_aspirasi' => 'required|string|max:255',
            'isi_aspirasi' => 'required|string',
            'lampiran.*' => 'nullable|file|mimes:pdf,doc,docx,jpg,jpeg,png|max:5120',
            'latitude' => 'nullable|numeric|between:-90,90',
            'longitude' => 'nullable|numeric|between:-180,180',
            'prioritas' => 'required|in:rendah,sedang,tinggi,urgent'
        ]);

        $data = $request->all();

        // Handle new file uploads
        if ($request->hasFile('lampiran')) {
            // Delete old files if needed
            if ($aspirasi->lampiran) {
                foreach ($aspirasi->lampiran as $file) {
                    Storage::disk('public')->delete($file['path']);
                }
            }

            $lampiran = [];
            foreach ($request->file('lampiran') as $file) {
                $filename = time() . '_' . $file->getClientOriginalName();

                $lampiran[] = [
                    'filename' => $file->getClientOriginalName(),
                    'path' => $path,
                    'size' => $file->getSize(),
                    'mime_type' => $file->getMimeType()
                ];
            }
            $data['lampiran'] = $lampiran;
        }

        $aspirasi->update($data);

        return redirect()->route('aspirasi.index')
                         ->with('success', 'Aspirasi berhasil diperbarui.');
    }

    public function destroy(Aspirasi $aspirasi)
    {
        try {
            // Delete associated files
            if ($aspirasi->lampiran) {
                foreach ($aspirasi->lampiran as $file) {
                    Storage::disk('public')->delete($file['path']);
                }
            }

            $aspirasi->delete();
            return redirect()->route('aspirasi.index')
                             ->with('success', 'Aspirasi berhasil dihapus.');
        } catch (\Exception $e) {
            return redirect()->route('aspirasi.index')
                             ->with('error', 'Terjadi kesalahan saat menghapus aspirasi.');
        }
    }

    // Method untuk update status

            'status' => 'required|in:pending,diproses,selesai,ditolak',
            'tanggapan_admin' => 'nullable|string'
        ]);

        $aspirasi->updateStatus(
            $request->status,
            auth()->id(),
            $request->tanggapan_admin
        );

        return redirect()->back()
                         ->with('success', 'Status aspirasi berhasil diperbarui.');
    }

    // Method untuk dashboard/statistik
    public function dashboard()
    {
        $stats = [
            'total' => Aspirasi::count(),
            'pending' => Aspirasi::pending()->count(),
            'diproses' => Aspirasi::diproses()->count(),
            'selesai' => Aspirasi::selesai()->count(),
            'bulan_ini' => Aspirasi::whereMonth('created_at', Carbon::now()->month)
                                 ->whereYear('created_at', Carbon::now()->year)
                                 ->count(),
        ];

        // Data untuk chart
        $chartData = Aspirasi::selectRaw('DATE(created_at) as tanggal, COUNT(*) as jumlah')
                           ->whereDate('created_at', '>=', Carbon::now()->subDays(30))
                           ->groupBy('tanggal')
                           ->orderBy('tanggal')
                           ->get();

        // Data berdasarkan kategori
        $byCategory = Aspirasi::with('categoriesAspirasi')
                            ->selectRaw('categories_aspirasi_id, COUNT(*) as jumlah')
                            ->groupBy('categories_aspirasi_id')
                            ->orderBy('jumlah', 'desc')
                            ->limit(5)
                            ->get();

        return view('aspirasi.dashboard', compact('stats', 'chartData', 'byCategory'));
    }

    // Method untuk download lampiran
    public function downloadLampiran(Aspirasi $aspirasi, $index)
    {
        if (!isset($aspirasi->lampiran[$index])) {
            abort(404);
        }

        $file = $aspirasi->lampiran[$index];
        $filePath = storage_path('app/public/' . $file['path']);

        if (!file_exists($filePath)) {
            abort(404);
        }

        return response()->download($filePath, $file['filename']);
    }

    public function publicStore(Request $request)
    {
        $request->validate([
            'kategori_aspirasi_id' => 'required|exists:kategori_aspirasis,id',
            'nama_pengirim' => 'required|string|max:255',
            'email' => 'required|email',
            'phone' => 'nullable|string|max:20',
            'alamat' => 'required|string',
            'jenis_aspirasi' => 'required|in:usulan,keluhan,kritik,saran',
            'judul_aspirasi' => 'required|string|max:255',
            'isi_aspirasi' => 'required|string',
            'lampiran.*' => 'nullable|file|mimes:pdf,doc,docx,jpg,jpeg,png|max:5120',
            'latitude' => 'nullable|numeric|between:-90,90',
            'longitude' => 'nullable|numeric|between:-180,180'
        ]);

        $data = $request->all();
        $data['prioritas'] = 'sedang'; // Default priority for public submissions

        // Handle file uploads
        if ($request->hasFile('lampiran')) {
            $lampiran = [];
            foreach ($request->file('lampiran') as $file) {
                $filename = time() . '_' . $file->getClientOriginalName();
                $path = $file->storeAs('aspirasi-lampiran', $filename, 'public');
                $lampiran[] = [
                    'filename' => $file->getClientOriginalName(),
                    'path' => $path,
                    'size' => $file->getSize(),
                    'mime_type' => $file->getMimeType()
                ];
            }
            $data['lampiran'] = $lampiran;
        }

        $aspirasi = Aspirasi::create($data);

        return redirect()->route('public.aspirasi.success')
                         ->with('success', 'Aspirasi berhasil dikirim dengan nomor tiket: ' . $aspirasi->nomor_tiket);
    }

    // Method untuk tracking aspirasi berdasarkan nomor tiket
    public function track(Request $request)
    {
        $aspirasi = null;
        
        if ($request->has('nomor_tiket') && $request->nomor_tiket != '') {
            $aspirasi = Aspirasi::with(['kategoriAspirasi.opd', 'admin'])
                              ->where('nomor_tiket', $request->nomor_tiket)
                              ->first();
        }

        return view('public.aspirasi.track', compact('aspirasi'));
    }
}
